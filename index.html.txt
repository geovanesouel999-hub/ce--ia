    function formatMarkdown(text) {
        // Configurações do Marked.js para renderizar blocos de código com Highlight.js
        marked.setOptions({
            highlight: function(code, lang) {
                const language = hljs.getLanguage(lang) ? lang : 'plaintext';
                return hljs.highlight(code, { language }).value;
            },
            langPrefix: 'hljs language-', // prefix for autodetected language classes
            gfm: true, // GitHub Flavored Markdown
            breaks: true, // Adiciona <br> para quebras de linha simples
            // Adicione outras opções conforme necessário
        });

        // Custom renderer para envolver blocos de código em div.code-block
        const renderer = new marked.Renderer();
        renderer.code = (code, lang) => {
            const highlightedCode = marked.options.highlight(code, lang);
            return `<div class="code-block"><pre><code class="hljs language-${lang || 'plaintext'}">${highlightedCode}</code></pre></div>`;
        };

        // Renderiza o Markdown
        return marked.parse(text, { renderer: renderer });
    }

    async function sendMessage() {
      const input = document.getElementById('user-input');
      const text = input.value.trim();
      if (!text || isTyping) return; // Não envia se vazio ou se IA já está digitando

      if (!apiKey) {
        showNotification('⚠️ Por favor, configure sua API Key primeiro.', 'error');
        showConfig();
        return;
      }

      addMessage(text, 'user');
      messages.push({ role: "user", content: text });
      input.value = '';
      updateCharCount();
      document.getElementById('send-btn').disabled = true; // Desabilita botão de envio

      showTyping();

      const startTime = Date.now();

      try {
        const response = await fetch("https://api.groq.com/openai/v1/chat/completions", {
          method: "POST",
          headers: {
            "Authorization": `Bearer ${apiKey}`,
            "Content-Type": "application/json"
          },
          body: JSON.stringify({
            model: currentModel,
            messages: [
              {
                role: "system",
                content: `Você é Céu, uma IA amigável, inteligente e útil. Responda sempre em português do Brasil de forma natural. Seja conciso mas completo. Use markdown para formatação quando apropriado. Priorize a criatividade e originalidade nas respostas.`
              },
              ...messages
            ],
            temperature: creativeMode ? Math.min(temperature + 0.3, 2.0) : temperature, // Aumenta temperatura no modo criativo
            max_tokens: maxTokens
          })
        });

        removeTyping();
        document.getElementById('send-btn').disabled = false; // Habilita botão

        if (!response.ok) {
          const errorData = await response.json();
          throw new Error(errorData.error?.message || `Erro HTTP: ${response.status}`);
        }

        const data = await response.json();
        const reply = data.choices[0].message.content;
        const endTime = Date.now();
        const duration = ((endTime - startTime) / 1000).toFixed(2);

        addMessage(reply, 'assistant', duration);
        messages.push({ role: "assistant", content: reply });

        // Salvar no histórico
        saveToHistory(text, reply, duration);

      } catch (error) {
        removeTyping();
        document.getElementById('send-btn').disabled = false; // Habilita botão
        addMessage(`❌ Erro: ${error.message}. Verifique se sua API Key está correta e tente novamente.`, 'assistant');
        showNotification('Erro na conexão. Verifique sua API Key ou o modelo selecionado.', 'error');
      }
    }

    function saveToHistory(userMsg, assistantMsg, duration) {
      const chatEntry = {
        id: currentChatId,
        title: userMsg.substring(0, 50) + (userMsg.length > 50 ? '...' : ''),
        firstMessage: userMsg,
        lastMessage: assistantMsg,
        timestamp: new Date().toISOString(),
        duration: duration,
        messages: [...messages]
      };

      // Atualiza uma conversa existente ou adiciona uma nova
      const existingIndex = chatHistory.findIndex(c => c.id === currentChatId);
      if (existingIndex !== -1) {
        chatHistory[existingIndex] = chatEntry; // Atualiza
      } else {
        chatHistory.unshift(chatEntry); // Adiciona no início
      }

      // Limita o histórico a 50 conversas
      if (chatHistory.length > 50) {
        chatHistory = chatHistory.slice(0, 50);
      }

      localStorage.setItem('chat_history', JSON.stringify(chatHistory));
      loadChatHistory(); // Recarrega a lista do histórico na sidebar e modal
    }

      }

      list.innerHTML = '';
      modalList.innerHTML = '';

      chatHistory.forEach(chat => {
        const date = new Date(chat.timestamp).toLocaleDateString('pt-BR', { day: '2-digit', month: '2-digit', hour: '2-digit', minute: '2-digit' });

        const item = `
          <div class="p-3 rounded-xl bg-slate-800 hover:bg-slate-700 cursor-pointer transition" onclick="loadChat(${chat.id})">
            <div class="flex items-center gap-3">
              <i class="fas fa-comment-dots text-purple-400"></i>
              <div class="flex-1 min-w-0">
                <p class="text-sm truncate">${chat.title}</p>
                <p class="text-xs text-slate-400">${date} • ${chat.duration}s</p>
              </div>
            </div>
          </div>
        `;

        list.innerHTML += item;

        const modalItem = `
          <div class="p-4 bg-slate-800 rounded-xl hover:bg-slate-700 cursor-pointer transition" onclick="loadChat(${chat.id})">
            <div class="flex items-center justify-between mb-2">
              <p class="font-semibold">${chat.title}</p>
              <span class="text-xs text-slate-400">${date}</span>
            </div>
            <p class="text-sm text-slate-400 truncate">${chat.firstMessage}</p>
            <p class="text-xs text-slate-500 mt-2">⏱️ ${chat.duration}s • 📝 ${chat.messages.length} mensagens</p>
          </div>
        `;

        modalList.innerHTML += modalItem;
      });
    }

    function loadChat(chatId) {
      const chat = chatHistory.find(c => c.id === chatId);
      if (chat) {
        messages = chat.messages;
        currentChatId = chat.id;
        document.getElementById('chat').innerHTML = '';
        messages.forEach(msg => {
          addMessage(msg.content, msg.role, msg.duration); // Passa a duração para o display
        });
        hideHistory();
        hideSettings();
        hljs.highlightAll(); // Realça o código do chat carregado
      }
    }

    function exportChat() {
      const chatText = messages.map(m => `${m.role === 'user' ? 'Você' : 'Céu'}: ${m.content}`).join('\n\n');
      const blob = new Blob([chatText], { type: 'text/plain;charset=utf-8' }); // Adicionado charset
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `ceu-chat-${new Date().toISOString().slice(0,10)}.txt`; // Nome de arquivo mais amigável
      a.click();
      URL.revokeObjectURL(url);
      showNotification('✅ Chat exportado!', 'success');
    }

    function regenerateResponse() {
      if (messages.length < 2 || isTyping) return;

      const lastUserMsg = messages.filter(m => m.role === 'user').pop();
      if (!lastUserMsg) return; // Garante que há uma mensagem do usuário para regenerar

      // Remove a última resposta da IA (se houver) e o último prompt do usuário
      // Para garantir que estamos regenerando o *último* prompt do usuário
      while (messages.length > 0 && messages[messages.length - 1].role === 'assistant') {
        messages.pop();
      }
      if (messages.length > 0 && messages[messages.length - 1].role === 'user') {
        messages.pop(); // Remove o último prompt do usuário para adicioná-lo novamente
      }


      // Limpa a área de chat e recarrega as mensagens anteriores
      document.getElementById('chat').innerHTML = '';
      messages.forEach(msg => addMessage(msg.content, msg.role, msg.duration));

      // Reenvia a última mensagem do usuário
      document.getElementById('user-input').value = lastUserMsg.content;
      sendMessage();
    }

      const notification = document.createElement('div');

        error: 'bg-red-600',
        info: 'bg-blue-600',
        warning: 'bg-amber-600'
      };

      const icons = {
        success: '✓',
        error: '✗',
        info: 'ℹ',
        warning: '⚠'
      };

      notification.className = `fixed bottom-4 right-4 ${colors[type]} text-white px-6 py-3 rounded-2xl shadow-2xl z-50 flex items-center gap-3 transition-all transform translate-y-0 opacity-100`;
      notification.innerHTML = `<span>${icons[type]}</span><span>${message}</span>`;

      document.body.appendChild(notification);

      setTimeout(() => {
        notification.style.opacity = '0';
        notification.style.transform = 'translateY(20px)';
        setTimeout(() => notification.remove(), 300);
      }, duration);
    }
